# Firestore rules suggestion (development -> tighten for production)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Users: each user can read/write their own document
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Courses: public read, restricted writes for admins (use custom claims 'admin')
    match /courses/{courseId} {
      allow read: if true;
      allow write: if false; // restrict root writes

      match /modules/{moduleId} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.token.admin == true;
      }
    }

    // Testimonials: public read and create, admins can update/delete
    match /testimonios/{testimonioId} {
      allow read: if true;
      allow create: if true;
      allow update: if request.data.diff(resource.data).affectedKeys().hasOnly(['likes', 'likedBy']) || (request.auth != null && request.auth.token.admin == true);
      allow delete: if request.auth != null && request.auth.token.admin == true;
    }

    // Default: deny
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
